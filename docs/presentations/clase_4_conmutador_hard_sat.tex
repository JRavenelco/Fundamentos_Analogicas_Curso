\begin{frame}{Diagnóstico del circuito anterior}
  \begin{alertblock}{Dos fallas eléctricas}
    \begin{enumerate}
      \item En SPICE, un BJT debe usar prefijo \texttt{Q}; \texttt{T} describe
        una línea de transmisión.
      \item Un NPN con emisor a GND y colector sin pull-up solo puede
        \textbf{hundir} corriente: no podía alimentar la base de T2.
    \end{enumerate}
  \end{alertblock}
  \begin{block}{Corrección}
    Q1 será un inversor/pull-down y una resistencia $R_{PU}$ desde $+5\,\mathrm{V}$
    suministrará la corriente de base de Q2.
  \end{block}
  \textbf{Predicción:} con entrada alta, ¿Q2 debe quedar encendido o apagado?
\end{frame}

\begin{frame}{3 · Dimensionar la base para el peor instante}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      Corriente de base aproximada:
      \[
        I_{B2}\approx\frac{5-0.8}{910}=\SI{4.6}{\milli\ampere}
      \]
      Corriente permanente desde $R_C$:
      \[
        I_{RC}\approx\frac{5-0.2}{220}=\SI{21.8}{\milli\ampere}
      \]
    \end{column}
    \begin{column}{0.45\textwidth}
      Al encender Q2 con $V_C\approx\SI{4.9}{\volt}$:
      \[
        I_{C,pico}\approx I_{RC}+\frac{V_C-0.2}{220}
        \approx\SI{43}{\milli\ampere}
      \]
      \[
        \beta_{forzado}\approx\frac{43}{4.6}\approx9.4
      \]
    \end{column}
  \end{columns}
  \begin{alertblock}{El error que se evita}
    Dimensionar solo $5/220$ omite la descarga simultánea del capacitor.
  \end{alertblock}
\end{frame}

\begin{frame}{4 · Hay dos constantes de tiempo}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Q2 OFF: carga}
        La trayectoria incluye $R_C+R_{ser}$:
        \[
          \tau_{\uparrow}=(220+220)\cdot\SI{5}{\micro\farad}
          =\SI{2.2}{\milli\second}
        \]
        \[
          V_C(t)=5\left(1-e^{-t/\tau_{\uparrow}}\right)
        \]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Q2 ON: descarga}
        La trayectoria principal incluye $R_{ser}$:
        \[
          \tau_{\downarrow}\approx220\cdot\SI{5}{\micro\farad}
          =\SI{1.1}{\milli\second}
        \]
        \[
          V_C(t)\approx V_C(0)e^{-t/\tau_{\downarrow}}
        \]
      \end{block}
    \end{column}
  \end{columns}
  \textbf{Predicción:} ¿qué transición debe verse dos veces más lenta?
\end{frame}

\begin{frame}{6 · Qué significa saturación forzada}
  \begin{itemize}
    \item Se diseña con una relación $I_C/I_B$ conservadora en lugar de usar el
      $\beta$ de región activa.
    \item La heurística ``10:1'' debe contrastarse con la tabla de
      $V_{CE(sat)}$ del transistor real para el par $I_C/I_B$ elegido.
    \item Beneficio estático: menor $V_{CE}$ y menor disipación de conducción.
    \item Compromiso dinámico: saturar profundo almacena carga y puede retrasar
      el apagado.
  \end{itemize}
  \begin{block}{Cuando importa la velocidad}
    Evaluar un clamp Baker/Schottky, reducir la sobreexcitación o usar un
    MOSFET adecuado; no afirmar que ``más saturación siempre es más rápida''.
  \end{block}
\end{frame}

\begin{frame}{7 · Sustituir Q2 por un MOSFET}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{MOSFET de nivel lógico}
        Se define por tener $R_{DS(on)}$ garantizada a la tensión lógica real
        disponible, por ejemplo 2.5, 3.3 o 4.5 V; no por un $V_{th}$ bajo.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{IRF640N}
        No es reemplazo directo para un gate-drive lógico de 5 V: su $R_{DS(on)}$ se
        especifica a $V_{GS}=\SI{10}{\volt}$. Requiere un driver de 10--12 V.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  El pico capacitivo y la disipación siguen existiendo. Cambia el
  dimensionamiento de base por el de carga de gate, $R_{DS(on)}$, SOA y térmica.
\end{frame}

\begin{frame}{8 · Seguridad y aplicación}
  \begin{itemize}
    \item Una carga capacitiva puede duplicar transitoriamente la corriente que
      sugeriría el pull-up por sí solo.
    \item Verificar corriente pico, potencia pulsada, SOA y repetición; no basta
      con la corriente permanente.
    \item En actuadores, buses auxiliares y líneas térmicas, documentar el estado
      seguro si el driver queda abierto o sin alimentación.
    \item En vacío, definir la ruta conductiva hacia estructura y el balance
      radiativo; no reutilizar sin análisis un $R_{\theta JA}$ de aire quieto.
  \end{itemize}
\end{frame}

  \end{enumerate}
\end{frame}
